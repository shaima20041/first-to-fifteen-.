<script>
let available, player, ai, gameOver;

function startGame(){
    available = [1,2,3,4,5,6,7,8,9];
    player = [];
    ai = [];
    gameOver = false;

    document.getElementById("board").innerHTML = "";
    document.getElementById("result").innerHTML = "";
    document.getElementById("restart").style.display = "none";

    createBoard();
    updateUI();
}

function createBoard(){
    for(let i=1;i<=9;i++){
        let btn = document.createElement("button");
        btn.innerText = i;
        btn.id = "btn"+i;
        btn.onclick = ()=>playerMove(i);
        document.getElementById("board").appendChild(btn);
    }
}

function updateUI(){
    document.getElementById("you").innerHTML =
        " Your Picks: " + player.join(" , ");

    document.getElementById("ai").innerHTML =
        " AI Picks: " + ai.join(" , ");
}

function has15(arr){
    if(arr.length < 3) return false;

    for(let i=0;i<arr.length;i++){
        for(let j=i+1;j<arr.length;j++){
            for(let k=j+1;k<arr.length;k++){
                if(arr[i]+arr[j]+arr[k] === 15){
                    return true;
                }
            }
        }
    }
    return false;
}

function disable(num){
    let btn = document.getElementById("btn"+num);
    btn.disabled = true;
}

function endGame(msg,color){
    gameOver = true;
    let result = document.getElementById("result");
    result.innerHTML = msg;
    result.style.color = color;

    document.getElementById("restart").style.display = "inline-block";
}

function playerMove(num){
    if(gameOver || !available.includes(num)) return;

    player.push(num);
    available = available.filter(x=>x!==num);
    disable(num);
    updateUI();

    if(has15(player)){
        endGame(" You Win!","lightgreen");
        return;
    }

    if(available.length===0){
        endGame(" Draw","white");
        return;
    }

    setTimeout(aiMove,700);
}

function aiMove(){
    if(gameOver) return;

    let pick = available[Math.floor(Math.random()*available.length)];

    ai.push(pick);
    available = available.filter(x=>x!==pick);
    disable(pick);
    updateUI();

    if(has15(ai)){
        endGame(" AI Wins!","red");
        return;
    }

    if(available.length===0){
        endGame(" Draw","white");
    }
}

function restartGame(){
    startGame();
}

startGame();
